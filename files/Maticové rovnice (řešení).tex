\section*{Maticové rovnice -- řešení}

Určete matici $X$, pro kterou platí: \quad $A\cdot X\cdot B=C$, \quad kde
\[
A=\begin{pmatrix} 2 & -3 & 1\\ 4 & -5 & 2\\ 5 & -7 & 3 \end{pmatrix},\quad
B=\begin{pmatrix} 9 & 7 & 6\\ 1 & 1 & 2\\ 1 & 1 & 1 \end{pmatrix},\quad
C=\begin{pmatrix} 2 & 0 & -2\\ 18 & 12 & 9\\ 23 & 15 & 11 \end{pmatrix}
\]
\hrule
\begin{align*}
A\cdot X\cdot B &= C \qquad \big/\cdot A^{-1}\ \text{zleva}\\
\underbrace{A^{-1}\cdot A}_{E}\cdot X\cdot B &= A^{-1}\cdot C \qquad \big/\cdot B^{-1}\ \text{zprava}\\
X\cdot\underbrace{B\cdot B^{-1}}_{E} &= A^{-1}\cdot C\cdot B^{-1} \qquad\Longrightarrow\qquad \pozn{X=A^{-1}\cdot C\cdot B^{-1}}
\end{align*}

\subsection*{\pozn{$A^{-1}$:}}
\begin{align*}
&\aug{2 & -3 & 1 & 1 & 0 & 0\\ 4 & -5 & 2 & 0 & 1 & 0\\ 5 & -7 & 3 & 0 & 0 & 1}
\ \krok{$R_2+(-2)R_1$}\ \sim
\aug{2 & -3 & 1 & 1 & 0 & 0\\ 0 & 1 & 0 & -2 & 1 & 0\\ 5 & -7 & 3 & 0 & 0 & 1}
\ \krok{$(-5)R_1+2R_3$}\ \sim\\
\sim{}&\aug{2 & -3 & 1 & 1 & 0 & 0\\ 0 & 1 & 0 & -2 & 1 & 0\\ 0 & 1 & 1 & -5 & 0 & 2}
\ \krok{$R_1+3R_2,\ R_3+(-1)R_2$}\ \sim
\aug{2 & 0 & 1 & -5 & 3 & 0\\ 0 & 1 & 0 & -2 & 1 & 0\\ 0 & 0 & 1 & -3 & -1 & 2}
\ \krok{$R_1+(-1)R_3$}\ \sim\\
\sim{}&\aug{2 & 0 & 0 & -2 & 4 & -2\\ 0 & 1 & 0 & -2 & 1 & 0\\ 0 & 0 & 1 & -3 & -1 & 2}
\ \krok{$\frac12R_1$}\ \sim
\left(\begin{array}{rrr|rrr} 1 & 0 & 0 & -1 & 2 & -1\\ 0 & 1 & 0 & -2 & 1 & 0\\ 0 & 0 & 1 & -3 & -1 & 2 \end{array}\right)
\ \Rightarrow\ \pozn{A^{-1}=\begin{pmatrix} -1 & 2 & -1\\ -2 & 1 & 0\\ -3 & -1 & 2 \end{pmatrix}}
\end{align*}

\subsection*{\pozn{$B^{-1}$:}}
\begin{align*}
&\aug{9 & 7 & 6 & 1 & 0 & 0\\ 1 & 1 & 2 & 0 & 1 & 0\\ 1 & 1 & 1 & 0 & 0 & 1}
\ \krok{$R_1\leftrightarrow R_3$}\ \sim
\aug{1 & 1 & 1 & 0 & 0 & 1\\ 1 & 1 & 2 & 0 & 1 & 0\\ 9 & 7 & 6 & 1 & 0 & 0}
\ \krok{$R_2+(-1)R_1,\ R_3+(-9)R_1$}\ \sim\\
\sim{}&\aug{1 & 1 & 1 & 0 & 0 & 1\\ 0 & 0 & 1 & 0 & 1 & -1\\ 0 & -2 & -3 & 1 & 0 & -9}
\ \krok{$R_2\leftrightarrow R_3$}\ \sim
\aug{1 & 1 & 1 & 0 & 0 & 1\\ 0 & -2 & -3 & 1 & 0 & -9\\ 0 & 0 & 1 & 0 & 1 & -1}
\ \krok{$R_1+(-1)R_3,\ R_2+3R_3$}\ \sim\\
\sim{}&\aug{1 & 1 & 0 & 0 & -1 & 2\\ 0 & -2 & 0 & 1 & 3 & -12\\ 0 & 0 & 1 & 0 & 1 & -1}
\ \krok{$\left(-\frac12\right)R_2$}\ \sim
\aug{1 & 1 & 0 & 0 & -1 & 2\\ 0 & 1 & 0 & -\frac12 & -\frac32 & 6\\ 0 & 0 & 1 & 0 & 1 & -1}
\ \krok{$R_1+(-1)R_2$}\ \sim\\
\sim{}&\aug{1 & 0 & 0 & \frac12 & \frac12 & -4\\ 0 & 1 & 0 & -\frac12 & -\frac32 & 6\\ 0 & 0 & 1 & 0 & 1 & -1}
\ \Rightarrow\ \pozn{B^{-1}=\frac12\begin{pmatrix} 1 & 1 & -8\\ -1 & -3 & 12\\ 0 & 2 & -2 \end{pmatrix}}
\end{align*}

\subsection*{Výpočet $X$:}
\[
A^{-1}\cdot C = \begin{pmatrix} -1 & 2 & -1\\ -2 & 1 & 0\\ -3 & -1 & 2 \end{pmatrix}\cdot
\begin{pmatrix} 2 & 0 & -2\\ 18 & 12 & 9\\ 23 & 15 & 11 \end{pmatrix}
= \begin{pmatrix} 11 & 9 & 9\\ 14 & 12 & 13\\ 22 & 18 & 19 \end{pmatrix}
\]
\[
A^{-1}\cdot C\cdot B^{-1} = \begin{pmatrix} 11 & 9 & 9\\ 14 & 12 & 13\\ 22 & 18 & 19 \end{pmatrix}\cdot\frac12\cdot
\begin{pmatrix} 1 & 1 & -8\\ -1 & -3 & 12\\ 0 & 2 & -2 \end{pmatrix}
= \frac12\cdot\begin{pmatrix} 2 & 2 & 2\\ 2 & 4 & 6\\ 4 & 6 & 2 \end{pmatrix}
\]
\[
\vysl{X=\begin{pmatrix} 1 & 1 & 1\\ 1 & 2 & 3\\ 2 & 3 & 1 \end{pmatrix}}
\]

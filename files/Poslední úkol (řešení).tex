\textbf{\textcircled{1}} Jsou dány body $A[1,2,2]$, $B[3,1,2]$, $C[5,1,1]$. Určete
\begin{enumerate}[label=\alph*), nosep]
\item rovnici roviny $\alpha=(A,B,C)$,
\item vektor $\vec x$ délky 12 kolmý k $\alpha$,
\item vzdálenost $D[4,-1,2]$ od $\alpha$.
\end{enumerate}
\hrule
\medskip

\textbf{a)}\quad $\vv{AB}=B-A=(2,-1,0)$, \qquad $\vv{AC}=C-A=(4,-1,-1)$
\[
\vv{AB}\times\vv{AC} = \begin{vmatrix} \vec i & \vec j & \vec k\\ 2 & -1 & 0\\ 4 & -1 & -1 \end{vmatrix} = (1,2,2)
\]
\[
\left.\begin{array}{l}
\alpha\colon x+2y+2z+d=0\\
A\in\alpha\colon\ 1+2\cdot2+2\cdot2+d=0 \ \Rightarrow\ d=-9
\end{array}\right\}\quad \vysl{\alpha\colon x+2y+2z-9=0}
\]

\textbf{b)}\quad $\vec x=t\cdot\vec n_\alpha=t(1,2,2)=(t,2t,2t)$
\[
12=\norm{\vec x}=\sqrt{t^2+(2t)^2+(2t)^2}=\sqrt{9t^2}=3t \ \Rightarrow\ t=4,\qquad \vysl{\vec x=(4,8,8)}
\]

\textbf{c)}
\begin{enumerate}[label=\arabic*), nosep]
\item bodem $D$ kolmice $k$ k $\alpha$: \quad $k\colon\ x=4+t,\ \ y=-1+2t,\ \ z=2+2t$
\item $k\cap\alpha$:
\begin{align*}
4+t+2(-1+2t)+2(2+2t)-9 &= 0\\
4+t-2+4t+4+4t-9 &= 0\\
9t &= 3\\
t &= \tfrac13 \qquad P\left[\tfrac{13}{3},-\tfrac13,\tfrac83\right]
\end{align*}
\item vzdálenost $D$ od $\alpha$ $=\norm{\vv{PD}}$:
\begin{align*}
\vv{PD} &= D-P = [4,-1,2]-\left[\tfrac{13}{3},-\tfrac13,\tfrac83\right] = \left(-\tfrac13,-\tfrac23,-\tfrac23\right)\\
\norm{\vv{PD}} &= \sqrt{\left(-\tfrac13\right)^2+\left(-\tfrac23\right)^2+\left(-\tfrac23\right)^2} = \sqrt{\tfrac99} = 1
\end{align*}
\pozn{vzdálenost $D$ od $\alpha$ je $\boxed{1}$}
\end{enumerate}

\textbf{\textcircled{3}} Vzdálenost rovnoběžných rovin $\alpha$ a $\beta$: \quad $\alpha\colon x-3z+2=0$, \quad $\beta\colon 2x-6z-7=0$
\hrule
\medskip

$\vec n_\alpha=\vec n_\beta=(1,0,-3)$
\begin{enumerate}[label=\arabic*), nosep]
\item $A\in\alpha$ libovolný \ldots\ $A\left[0,0,\frac23\right]$
\item $k\perp\alpha$, $A\in k$ \ldots\ $k=A+t\,\vec n_\alpha$: \quad $x=t,\ \ y=0,\ \ z=\frac23-3t$
\item $k\cap\beta=B$:
\begin{align*}
2t-6\left(\tfrac23-3t\right)-7 &= 0\\
2t-4+18t-7 &= 0\\
20t &= 11 \ \Rightarrow\ t=\tfrac{11}{20}
\end{align*}
$B\left[\frac{11}{20},0,\frac23-\frac{33}{20}\right]$
\item \pozn{vzdálenost $\alpha$ a $\beta$} $=\norm{\vv{AB}}$:
\begin{align*}
\vv{AB} &= B-A = \left[\tfrac{11}{20},0,\tfrac23-\tfrac{33}{20}\right]-\left[0,0,\tfrac23\right] = \left(\tfrac{11}{20},0,-\tfrac{33}{20}\right)\\
\norm{\vv{AB}} &= \sqrt{\left(\tfrac{11}{20}\right)^2+0^2+\left(-\tfrac{33}{20}\right)^2} = \sqrt{\tfrac{1210}{400}} = \sqrt{\tfrac{121}{40}} = \vysl{\frac{11}{2\sqrt{10}}}
\end{align*}
\end{enumerate}

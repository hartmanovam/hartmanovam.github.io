\section*{Vypočti:}

\[
\iint_M\frac{x^2}{1+y^2}\,\mathrm{d}M, \qquad M=\underbrace{\langle0,2\rangle}_{\text{\color{green!50!black}\uv{$x$}}}\times\underbrace{\langle0,1\rangle}_{\text{\color{red!75!black}\uv{$y$}}}
\]

\begin{align*}
{\color{green!50!black}\int_0^2}\left({\color{red!75!black}\int_0^1}\frac{x^2}{1+y^2}\,{\color{red!75!black}\mathrm{d}y}\right){\color{green!50!black}\mathrm{d}x}
&= \int_0^2\left(x^2\int_0^1\frac{1}{1+y^2}\dy\right)\dx =\\
&= \int_0^2\Big[x^2\arctg y\Big]_{y=0}^{y=1}\dx
= \int_0^2\Big(x^2\underbrace{\arctg1}_{45^\circ=\frac\pi4}-x^2\underbrace{\arctg0}_{0^\circ=0}\Big)\dx
=\\
&= \int_0^2\left(x^2\,\frac\pi4-0\right)\dx = \left[\frac\pi4\,\frac{x^3}{3}\right]_0^2 = \frac{8\pi}{12}-0 = \vysl{\frac{2\pi}{3}}
\end{align*}
({\color{red!75!black}meze pro $y$} jsou u vnitřního integrálu, {\color{green!50!black}meze pro $x$} u vnějšího.)

\textbf{2.}\quad
$\boxed{1\le x\le e}$ \quad \pozn{zde jsou meze pouze čísla, proto TO MUSÍM dát vně!}

\qquad\ \, $0\le y\le\ln x$

\begin{align*}
\int_1^e\left(\int_0^{\ln x}y\dy\right)\dx &= \int_1^e\left[\frac{y^2}{2}\right]_0^{\ln x}\dx
= \int_1^e\left(\frac{\ln^2x}{2}-0\right)\dx \overset{*}{=} \left[\frac12x\ln^2x-x\ln x+x\right]_1^e =\\
&= \frac12e\underbrace{(\ln e)^2}_{1}-e\underbrace{\ln e}_{1}+e-\Big(\frac12\cdot1\underbrace{(\ln1)^2}_{0}-1\cdot\underbrace{\ln1}_{0}+1\Big)
= \frac12e-\cancel{e}+\cancel{e}-1 = \vysl{\frac e2-1}
\end{align*}

\begin{tcolorbox}[colframe=blue!60!black, colback=white, coltitle=white, fonttitle=\bfseries, title={$*$ Pomocný výpočet:}, sharp corners]
\begin{align*}
\int\frac{(\ln x)^2}{2}\dx &= \frac12\int(\ln x)^2\cdot1\dx =
\left|\begin{array}{ll} u=(\ln x)^2 & \to\ u'=2\ln x\cdot\frac1x\\ v'=1 & \to\ v=x\end{array}\right| =\\
&= \frac12\left(\ln^2x\cdot x-\int2\ln x\,\frac{1}{\cancel{x}}\cdot\cancel{x}\dx\right) =
\left|\begin{array}{ll} u=\ln x & \to\ u'=\frac1x\\ v'=1 & \to\ v=x\end{array}\right| =\\
&= \frac12\left(x\ln^2x-2\cdot\left(\ln x\cdot x-\int\frac{1}{\cancel{x}}\cdot\cancel{x}\dx\right)\right)
= \frac12x\ln^2x-x\ln x+x
\end{align*}
\end{tcolorbox}

\section*{Zaměň pořadí integrace}

\[
\int_0^1\int_y^{\sqrt y}f(x,y)\dx\dy = \int_{y=0}^{y=1}\left(\int_{x=y}^{x=\sqrt y}f(x,y)\dx\right)\dy
\]

\begin{minipage}[t]{0.5\textwidth}
\fbox{\parbox{3.2cm}{Pás pro $y$:\\ $y=1$\\ $y=0$}}
\quad$\Rightarrow$\quad
\fbox{\parbox{3.2cm}{Pás pro $x$:\\ $x=0$\\ $x=1$}}
\end{minipage}%
\begin{minipage}[t]{0.5\textwidth}
Meze pro $x$ v opačném pořadí:
\begin{itemize}[nosep]
\item $x=y \ \longrightarrow\ \boxed{y=x}$
\item $x=\sqrt y\ \big/^2,\quad x^2=y \ \longrightarrow\ \boxed{y=x^2}$
\end{itemize}
\end{minipage}

\medskip
\textbf{1. obrázek}

\begin{minipage}{0.58\textwidth}
\begin{tikzpicture}
\begin{axis}[axis lines=middle, width=7.5cm, height=7cm, clip=false,
  xmin=-1, xmax=2, ymin=-0.6, ymax=2, xtick={1}, ytick={1}]
\addplot[name path=f, thick, blue!60!black, domain=-1.3:1.35, samples=80] {x^2} node[above] {$y=x^2$};
\addplot[name path=g, thick, blue!60!black, domain=-0.6:1.8] {x} node[right] {$y=x$};
\addplot[pattern=north east lines] fill between[of=f and g, soft clip={domain=0:1}];
\addplot[domain=-1:1.8] {1} node[right] {$y=1$};
\draw[dotted] (axis cs:1,0) -- (axis cs:1,1);
\node[right] at (axis cs:2,0) {$y=0$};
\end{axis}
\end{tikzpicture}
\end{minipage}%
\begin{minipage}{0.4\textwidth}
Průsečík $f=g$:
\begin{align*}
x &= x^2\\
x^2-x &= 0\\
x(x-1) &= 0\\
x=0,&\quad x=1
\end{align*}
\end{minipage}

\[
\boxed{\begin{array}{c} 0\le x\le1\\ x^2\le y\le x \end{array}}
\quad\Longrightarrow\quad
\underbrace{\boxed{\int_0^1\left(\int_{x^2}^{x}f(x,y)\dy\right)\dx}}_{\text{VÝSLEDEK}}
\]

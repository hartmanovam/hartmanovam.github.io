\textbf{4)}
\begin{align*}
\int\frac{-3}{(x+5)^2}\dx &= -3\int\frac{1}{(x+5)^2}\dx = \subst{x+5=t\\ \mathrm{d}x=\mathrm{d}t}
= -3\int\frac{1}{t^2}\dt = -3\int t^{-2}\dt =\\
&= -3\,\frac{t^{-1}}{-1}+c = +3\,\frac1t+c = \vysl{\frac{3}{x+5}+c}
\end{align*}

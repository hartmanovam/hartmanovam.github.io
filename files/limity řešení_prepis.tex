\section*{Vypočtěte:}

\textbf{a)}
\[
\lim_{x\to\infty}e^{\frac{2x}{x^2-1}} = e^0 = \vysl{1} \qquad \pozn{(složená funkce)}
\]
\[
\left(\lim_{x\to\infty}\frac{2x}{x^2-1} = \lim_{x\to\infty}\frac{x\cdot2}{x\left(x-\frac1x\right)} = \frac2\infty = 0\right)
\]

\textbf{b)}
\[
\lim_{x\to\infty}\frac{3x^4-2}{2x^4+x+3} = \lim_{x\to\infty}\frac{\cancel{x^4}\Big(3-\overbrace{\tfrac{2}{x^4}}^{\to0}\Big)}{\cancel{x^4}\Big(2+\underbrace{\tfrac{1}{x^3}}_{\to0}+\underbrace{\tfrac{3}{x^4}}_{\to0}\Big)} = \vysl{\frac32}
\]

\textbf{c)}
\[
\lim_{x\to-3^+}\frac{2-x}{2x^2+5x-3} = \left[\frac50\right] = \lim_{x\to-3^+}\frac{2-x}{2(x+3)\left(x-\frac12\right)} =
\]
\[
D=25-4\cdot2\cdot(-3), \quad D=49 \ \Rightarrow\ x_{1,2}=\frac{-5\pm7}{4}=\begin{cases}-3\\ \frac24=\frac12\end{cases}
\]
\[
= \lim_{x\to-3^+}\frac{1}{x+3}\cdot\lim_{x\to-3^+}\frac{2-x}{2\left(x-\frac12\right)} = \frac{1}{0^+}\cdot\frac{5}{2\left(-3-\frac12\right)} =
\]
\[
= \infty\cdot\frac{5}{2\cdot\left(-\frac72\right)} = \infty\cdot\left(-\frac57\right) = \vysl{-\infty}
\qquad \priblizeni{-3}{+}{(x+3)>0}
\]

\newpage
\textbf{d)}
\[
\lim_{x\to4^+}\ln\frac{3x+1}{x^2-4x} = \ln\text{\uv{$\infty$}} = \infty \qquad \Big(\lim_{x\to\infty}\ln x=\infty\Big)
\]
\[
\lim_{x\to4^+}\frac{3x+1}{x^2-4x} = \lim_{x\to4^+}\frac{3x+1}{x(x-4)} = \lim_{x\to4^+}\frac{1}{x-4}\cdot\lim\frac{3x+1}{x} =
\]
\[
= \frac{1}{0^+}\cdot\frac{3\cdot4+1}{4} = \infty\cdot\frac{13}{4} = \vysl{\infty}
\qquad \priblizeni{4}{}{(x-4)>0}
\]

\textbf{e)}
\begin{align*}
\lim_{x\to-\infty}\left(x^3-x+2\right) &= \lim_{x\to-\infty}x\Big(x^2-1+\underbrace{\tfrac2x}_{\to0}\Big) =\\
&= -\infty\cdot(-\infty)^2 = -\infty\cdot(+\infty) = \vysl{-\infty}
\end{align*}

\pozn{\textbf{!!}} \quad
